\[

  \to
  (K^{\rm bdry}_J,L_J,V_J,\langle\cdot,\cdot\rangle_J,P_J)
  \to
  K^{\rm eff}_J(\lambda)

  \lambda&-\sqrt J\\
  -\sqrt J&\lambda+J
